cmake_minimum_required(VERSION 3.20)
project(LocalFlowWhisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE ON CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE}" whisper)
add_executable(localflow-whisper worker.cpp)
target_link_libraries(localflow-whisper PRIVATE whisper)
target_compile_options(localflow-whisper PRIVATE -Wall -Wextra -Werror)
